% The font families this image installs so that they can be selected by name, as
% Overleaf documents them
% (https://www.overleaf.com/learn/latex/Questions/Which_OTF_or_TTF_fonts_are_supported_via_fontspec%3F).
%
% fontspec resolves a family name through fontconfig, so this example fails to
% compile when one of these families is missing - which is exactly the error it
% exists to catch ("The font ... cannot be found"). It is compiled with XeLaTeX
% (see the latexmkrc next to it) because that is the engine that goes through
% fontconfig.
%
% Every name below was read out of the font files of the package that provides
% it, not derived from the package name: `fonts-kacst` for example has no
% `KacstOne` (that is `fonts-kacst-one`), the Thai packages provide `Tlwg Typo`
% and not `TLWG Typo`, and `fonts-khmeros` has no plain `Khmer OS`.
%
% Add a family here when the image starts installing it, and remove one when it
% stops: this file is the list of what the image promises.

\usepackage{fontspec}

\newcommand{\showfont}[1]{%
  \par\noindent\textbf{#1}\quad{\fontspec{#1}The quick brown fox --- 0123456789}%
}
